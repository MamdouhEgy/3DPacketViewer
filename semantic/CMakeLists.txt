cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(StatefulSemanticProtocolAnalyzer VERSION 0.1.0 LANGUAGES CXX)
    enable_testing()
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Widgets Network Test)
add_library(sspa_core STATIC core/Model.cpp core/Engine.cpp core/Statistics.cpp core/Context.cpp assets/Policy.cpp extraction/Normalizer.cpp
    protocols/Iec104.cpp protocols/Goose.cpp protocols/Sv.cpp protocols/Mms.cpp protocols/Modbus.cpp protocols/Dnp3.cpp)
set_target_properties(sspa_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(sspa_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(sspa_core PUBLIC Qt6::Core)
if(NOT MSVC)
    target_compile_options(sspa_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(TARGET epan)
    add_library(sspa_extraction STATIC wireshark/Extractor.cpp)
    set_target_properties(sspa_extraction PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_link_libraries(sspa_extraction PUBLIC sspa_core epan)
endif()
add_library(sspa_serialization STATIC ai/Sanitizer.cpp ai/Contracts.cpp)
set_target_properties(sspa_serialization PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(sspa_serialization PUBLIC sspa_core)
add_library(sspa_ai STATIC ai/Provider.cpp)
set_target_properties(sspa_ai PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(sspa_ai PUBLIC sspa_serialization Qt6::Network)
if(TARGET epan)
    add_executable(semantic_analyze tools/AnalyzeCapture.cpp)
    target_link_libraries(semantic_analyze PRIVATE sspa_extraction sspa_serialization wiretap wsutil)
endif()
add_library(sspa_ai_ui STATIC ui/AiPanel.cpp)
set_target_properties(sspa_ai_ui PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(sspa_ai_ui PUBLIC sspa_ai Qt6::Widgets)
if(TARGET epan AND TARGET uiqt_plugin)
    include(WiresharkPlugin)
    get_directory_property(semantic_definitions COMPILE_DEFINITIONS)
    list(FILTER semantic_definitions EXCLUDE REGEX "^PLUGIN_VERSION=")
    set_directory_properties(PROPERTIES COMPILE_DEFINITIONS "${semantic_definitions}")
    set_module_info(semantic_analyzer 0 1 0 0)
    set(PLUGIN_FILES plugin.c wireshark/Plugin.cpp wireshark/Adapter.cpp ui/Viewer.cpp ui/Tables.cpp)
    register_plugin_files(plugin.c plugin_ui wireshark/Plugin.cpp)
    set_source_files_properties(plugin.c PROPERTIES SKIP_AUTOGEN ON)
    add_wireshark_plugin_library(semantic_analyzer ui)
    target_link_libraries(semantic_analyzer sspa_extraction sspa_ai_ui epan uiqt_plugin)
    install_plugin(semantic_analyzer ui)
endif()
add_executable(semantic_core_tests tests/unit/CoreTests.cpp)
target_link_libraries(semantic_core_tests PRIVATE sspa_ai Qt6::Test)
add_test(NAME semantic_core COMMAND semantic_core_tests)
option(SSPA_GUI_TESTS "Compile opt-in native semantic GUI validation driver" OFF)
if(SSPA_GUI_TESTS AND TARGET semantic_analyzer)
    target_sources(semantic_analyzer PRIVATE tests/integration/GuiTests.cpp)
    target_compile_definitions(semantic_analyzer PRIVATE SSPA_GUI_TESTS)
    target_link_libraries(semantic_analyzer Qt6::Test)
endif()
add_executable(semantic_network_tests tests/unit/NetworkTests.cpp)
target_link_libraries(semantic_network_tests PRIVATE sspa_ai Qt6::Test)
add_test(NAME semantic_network COMMAND semantic_network_tests)

if(NOT MSVC)
    foreach(target sspa_extraction sspa_serialization sspa_ai sspa_ai_ui semantic_analyzer semantic_analyze semantic_core_tests semantic_network_tests)
        if(TARGET ${target})
            target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        endif()
    endforeach()
endif()
