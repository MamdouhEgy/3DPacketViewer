cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(3DPacketViewer VERSION 0.1.0 LANGUAGES CXX)
    enable_testing()
endif()
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets OpenGL OpenGLWidgets Test)
add_library(packetviewer_core STATIC
    src/core/PacketModel.cpp src/core/FieldRangeResolver.cpp
    src/core/PacketGeometryEngine.cpp src/core/SelectionController.cpp
    src/rendering/CameraController.cpp)
set_target_properties(packetviewer_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(packetviewer_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(packetviewer_core PUBLIC Qt6::Core Qt6::Gui)
if(NOT MSVC)
    target_compile_options(packetviewer_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_library(packetviewer_ui STATIC src/rendering/PacketRenderer.cpp src/rendering/shaders.qrc
    src/ui/FieldInspector.cpp src/ui/RawByteView.cpp src/ui/PacketViewerWidget.cpp)
set_target_properties(packetviewer_ui PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(packetviewer_ui PUBLIC packetviewer_core Qt6::Widgets Qt6::OpenGL Qt6::OpenGLWidgets)
if(TARGET epan AND TARGET uiqt_plugin)
    include(WiresharkPlugin)
    set_module_info(packetviewer3d 0 1 0 0)
    set(PLUGIN_FILES plugin.c src/wireshark/Plugin.cpp src/wireshark/WiresharkAdapter.cpp src/wireshark/ProtoTreeExtractor.cpp)
    register_plugin_files(plugin.c plugin_ui src/wireshark/Plugin.cpp)
    set_source_files_properties(plugin.c PROPERTIES SKIP_AUTOGEN ON)
    add_wireshark_plugin_library(packetviewer3d ui)
    target_link_libraries(packetviewer3d packetviewer_ui epan uiqt_plugin)
    install_plugin(packetviewer3d ui)
endif()
add_executable(packetviewer_core_tests tests/unit/CoreTests.cpp)
target_link_libraries(packetviewer_core_tests PRIVATE packetviewer_core Qt6::Test)
add_test(NAME packetviewer_core COMMAND packetviewer_core_tests)
if(TARGET epan)
    add_executable(packetviewer_extract tests/integration/ExtractCapture.cpp src/wireshark/ProtoTreeExtractor.cpp)
    target_link_libraries(packetviewer_extract PRIVATE packetviewer_core epan wiretap wsutil)
endif()

option(PACKETVIEWER_GUI_TESTS "Compile opt-in in-process GUI test driver (development only)" OFF)
if(PACKETVIEWER_GUI_TESTS AND TARGET packetviewer3d)
    target_sources(packetviewer3d PRIVATE tests/integration/GuiTests.cpp)
    target_compile_definitions(packetviewer3d PRIVATE PACKETVIEWER_GUI_TESTS)
    target_link_libraries(packetviewer3d Qt6::Test)
endif()
